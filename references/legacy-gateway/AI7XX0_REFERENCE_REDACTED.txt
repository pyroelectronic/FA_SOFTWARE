# Function -> wait for dpkg done
wait_for_dpkg_lock() {
  local MAX_WAIT=300
  local WAIT=0

  while sudo fuser /var/lib/dpkg/lock-frontend >/dev/null 2>&1; do
    echo "dpkg ist gesperrt... warte (aktuell $WAIT Sekunden)"
    sleep 5
    WAIT=$((WAIT+5))
    if [ "$WAIT" -ge "$MAX_WAIT" ]; then
      echo "Timeout: dpkg bleibt gesperrt. Abbruch."
      exit 1
    fi
  done
}

# Function -> Safe update an upgrade the OS
safe_upgrade(){

# Auf APT-Lock warten
MAX_WAIT=300  # max. 5 Minuten warten
WAIT=0

while sudo fuser /var/lib/dpkg/lock-frontend >/dev/null 2>&1; do
  echo "APT ist gesperrt... warte (aktuell $WAIT Sekunden)"
  sleep 5
  WAIT=$((WAIT+5))
  if [ "$WAIT" -ge "$MAX_WAIT" ]; then
    echo "Timeout: APT bleibt gesperrt. Abbruch."
    exit 1
  fi
done

echo "APT ist frei - fahre mit Update fort."

apt-mark hold nvidia-l4t-kernel
apt-mark hold nvidia-l4t-kernel-dtbs
apt-mark hold nvidia-l4t-kernel-headers
apt-mark hold nvidia-l4t-kernel-oot-headers
apt-mark hold nvidia-l4t-kernel-oot-modules
apt-mark hold nvidia-l4t-bootloader
apt-mark hold nvidia-l4t-initrd
apt-get update -y
apt-get upgrade -y

echo "Safe update upgrade done"

}
